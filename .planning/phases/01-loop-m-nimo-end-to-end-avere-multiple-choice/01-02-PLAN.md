// src/screens/session.js
export function sessionScreen(content, state): AlpineComponent {
  ready: boolean,
  exerciseIds: string[],
  cursor: number,
  results: Array<{exerciseId, correct}>,
  selectedIndex: number | null,
  feedback: 'correct' | 'incorrect' | null,
  autoAdvanceHandle: number | null,

  init(): void,
  get currentExercise(): Exercise,
  get progressLabel(): string,  // "Ejercicio X / N"
  get done(): boolean,
  selectOption(idx: number): void,  // grades + sets feedback + schedules auto-advance if correct
  advance(): void,                  // cancels timer, increments cursor; if done, applies+saves state

  destroy(): void                   // Alpine hook; cancels pending timer
}
